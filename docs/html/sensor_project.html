121    t_known_s: ArrayLike, y_known: ArrayLike, t_query_s: ArrayLike

123    """Evaluate piecewise-linear interpolation with a readable Python loop.
124
125    Parameters
126    ----------
127    t_known_s : array_like, shape (n,)
128        Finite, strictly increasing known times in seconds, with ``n >= 2``.
129    y_known : array_like, shape (n,)
130        Finite retained observations in the sensor's value unit.
131    t_query_s : array_like, shape (m,)
132        Finite query times in the closed known interval, in any order.
133
134    Returns
135    -------
136    numpy.ndarray, shape (m,)
137        Piecewise-linear predictions in query order and the original value unit.
138
139    Raises
140    ------
141    ValueError
142        If observations or queries violate the same guards as :func:`reconstruct`.
143
144    Examples
145    --------
146    >>> scalar_linear([0, 2, 5], [400, 420, 450], [1, 3]).tolist()
147    [410.0, 430.0]
148
149    Notes
150    -----
151    This is a genuine scalar implementation of the same interpolation
152    algorithm, using binary segment search rather than a deliberately wasteful
153    scan. Comparing it with the vectorised implementation measures the cost of
154    Python-level iteration, not a different scientific method.
155    """
156    time, values = _series(t_known_s, y_known)
157    query = _queries(time, t_query_s)
158    time_list = time.tolist()
159    value_list = values.tolist()
160    prediction = np.empty(query.size, dtype=np.float64)
161    for index, requested_time in enumerate(query):
162        right = bisect_right(time_list, float(requested_time))
163        left = min(right - 1, len(time_list) - 2)
164        fraction = (requested_time - time_list[left]) / (time_list[left + 1] - time_list[left])
165        prediction[index] = value_list[left] + fraction * (value_list[left + 1] - value_list[left])
166    return prediction

176    """Remove one contiguous block and return only its neighbouring context.
177
178    Parameters
179    ----------
180    t_s : array_like, shape (n,)
181        Finite, strictly increasing full observation times in seconds.
182    y : array_like, shape (n,)
183        Matching finite sensor values in their original unit.
184    start : int
185        Zero-based index of the first held-out observation.
186    count : int
187        Positive number of consecutive observations to withhold.
188    context_points : int, default 4
189        Number of retained observations on each side. At least two are required
190        so a cubic method can estimate local slopes from neighbouring intervals.
191
192    Returns
193    -------
194    known_t_s : numpy.ndarray, shape (2 * context_points,)
195        Left then right retained times, strictly excluding the withheld block.
196    known_y : numpy.ndarray, shape (2 * context_points,)
197        Retained values corresponding to ``known_t_s``.
198    query_t_s : numpy.ndarray, shape (count,)
199        Times of the withheld observations, used as reconstruction queries.
200    truth : numpy.ndarray, shape (count,)
201        Withheld recorded values for scoring only, never an interpolator input.
202
203    Raises
204    ------
205    ValueError
206        If the full series is invalid, an index or count is not an integer, the
207        count is non-positive, fewer than two context points are requested per
208        side, or the block has insufficient neighbouring observations.
209
210    Examples
211    --------
212    >>> known_t, known_y, query_t, truth = mask_block(
213    ...     np.arange(10), 400 + np.arange(10), start=4, count=2, context_points=2
214    ... )
215    >>> known_t.tolist(), query_t.tolist(), truth.tolist()
216    ([2.0, 3.0, 6.0, 7.0], [4.0, 5.0], [404.0, 405.0])
217
218    Notes
219    -----
220    All outputs are copies. No withheld value is used to select or construct
221    context. The caller must select block locations and method settings without
222    consulting held-out reconstruction errors.
223    """
224    time, values = _series(t_s, y)
225    for name, number in (("start", start), ("count", count), ("context_points", context_points)):
226        if isinstance(number, (bool, np.bool_)) or not isinstance(number, Integral):
227            raise ValueError(f"{name} must be an integer")
228    if count < 1:
229        raise ValueError("count must be positive")
230    if context_points < 2:
231        raise ValueError("context_points must be at least two per side")
232    stop = start + count
233    if start < context_points or stop + context_points > time.size:
234        raise ValueError("the held-out block must have enough context observations on both sides")
235    known_time = np.concatenate((time[start - context_points:start], time[stop:stop + context_points]))
236    known_values = np.concatenate((values[start - context_points:start], values[stop:stop + context_points]))
237    return known_time, known_values, time[start:stop].copy(), values[start:stop].copy()

12def error_metrics(y_true: ArrayLike, y_pred: ArrayLike) -> dict[str, float]:
13    """Summarise reconstruction residuals against withheld recorded values.
14
15    Parameters
16    ----------
17    y_true : array_like, shape (n,)
18        At least one finite withheld recorded value. These are recorded targets,
19        not independently calibrated physical ground truth.
20    y_pred : array_like, shape (n,)
21        Matching finite reconstructed values in the same unit as ``y_true``.
22
23    Returns
24    -------
25    dict[str, float]
26        ``mae`` (mean absolute error), ``rmse`` (root mean squared error),
27        ``max_abs_error`` (largest absolute residual), and ``bias`` (mean of
28        prediction minus target). All quantities preserve the original unit.
29
30    Raises
31    ------
32    ValueError
33        If inputs are empty, non-finite, complex, non-vector, mismatched, or
34        produce residuals or metrics outside finite floating-point range.
35
36    Examples
37    --------
38    >>> error_metrics([400, 410, 420], [401, 408, 420])
39    {'mae': 1.0, 'rmse': 1.2909944487358056, 'max_abs_error': 2.0, 'bias': -0.3333333333333333}
40
41    Notes
42    -----
43    The square root of the mean squared error uses scaled residuals to avoid
44    an unnecessary overflow when finite sensor residuals are large. A positive
45    bias means that reconstruction overestimates the withheld measurements.
46    """
47    target = _finite_vector(y_true, "y_true", minimum=1)
48    prediction = _finite_vector(y_pred, "y_pred", minimum=1)
49    if target.size != prediction.size:
50        raise ValueError("y_true and y_pred must have the same length")
51    with np.errstate(over="ignore", invalid="ignore"):
52        residual = prediction - target
53    if not np.all(np.isfinite(residual)):
54        raise ValueError("residuals must remain within finite floating-point range")
55    scale = float(np.max(np.abs(residual)))
56    if scale == 0.0:
57        return {"mae": 0.0, "rmse": 0.0, "max_abs_error": 0.0, "bias": 0.0}
58    # Scaling also keeps mean reductions finite when the sum itself is huge.
59    scaled = residual / scale
60    result = {
61        "mae": float(scale * np.mean(np.abs(scaled))),
62        "rmse": float(scale * np.sqrt(np.mean(scaled * scaled))),
63        "max_abs_error": scale,
64        "bias": float(scale * np.mean(scaled)),
65    }
66    if not all(np.isfinite(number) for number in result.values()):
67        raise ValueError("error metrics must remain within finite floating-point range")
68    return result

246    inventory = [{"name": p.name, "bytes": p.stat().st_size, "sha256": _hash(p)} for p in required]
247    example_inventory = []
248    example_dir = source_dir / "Example Data (Do not use)"
249    if example_dir.is_dir():
250        for path in sorted(example_dir.rglob("*")):
251            if not path.is_file():
252                continue
253            item = {"name": str(path.relative_to(source_dir)), "bytes": path.stat().st_size, "excluded_from_analysis": True}
254            if path.suffix.lower() == ".csv":
255                with path.open(encoding="utf-8-sig", errors="replace", newline="") as stream:
256                    first = next(csv.reader(stream), [])
257                item["first_row_only"] = first
258                item["first_row_columns"] = len(first)
259                item["row_count"] = "not read: excluded example"
260            elif path.suffix.lower() == ".png":
261                from PIL import Image
262                with Image.open(path) as image:
263                    item["image_dimensions_px"] = list(image.size)
264            example_inventory.append(item)

266    environmental = _environment(source_dir / ENVIRONMENT_ZIP, env_locations, output_dir / "environment.csv")
267    occupancy = _occupancy(source_dir / OCCUPANCY_ZIP, occ_locations, output_dir / "occupancy_summary.csv")
268    audit = {"sources": inventory, "excluded_examples": example_inventory, "workbook": workbook,
269             "environment": environmental, "occupancy": occupancy, "primary_sensor_id": PRIMARY_SENSOR_ID,
270             "analysis_scope": "Retrospective artificial-gap CO2 reconstruction; not real-time forecasting, physical ventilation validation or occupancy inference."}
271    (output_dir / "source_audit.json").write_text(json.dumps(audit, indent=2, ensure_ascii=False), encoding="utf-8")
272    return audit

Stream both supplied ZIPs and produce compact, auditable prepared data.

Parameters
  • source_dir (pathlib.Path): Folder containing both original ZIPs and the sensor-location workbook.
  • output_dir (pathlib.Path): Folder for environment.csv, occupancy_summary.csv and source_audit.json. Existing files with those names are regenerated.
Returns
  • dict: Source hashes, workbook contents, cleaning decisions and stream counts.
Raises
  • FileNotFoundError: If any required supplied source is missing.
  • ValueError: If the source schema, units, finite values or duplicate consistency fail.
Notes

The excluded example folder is inventoried without analysis. Individual occupancy records are never exported. No room mapping is guessed.
